// Fetch all public repos (paginated, 100 per page)
// allRepos and langColour are used by Tasks 4 and 5 — do not redefine them.
allRepos = {
  const repos = [];
  let page = 1;
  while (true) {
    const r = await fetch(
      `https://api.github.com/orgs/Public-Health-Scotland/repos` +
      `?per_page=100&page=${page}&type=public&sort=pushed`,
      { headers: { "Accept": "application/vnd.github+json" } }
    );
    if (!r.ok) break;
    const batch = await r.json();
    if (!Array.isArray(batch) || batch.length === 0) break;
    repos.push(...batch);
    page++;
  }
  return repos;
}

langColour = ({
  R:          "#2166ac",
  Python:     "#3572A5",
  Vue:        "#42b883",
  JavaScript: "#f1e05a",
  TypeScript: "#3178c6",

// Returns #1b1b1b or #ffffff depending on which achieves better contrast
// against the given hex background colour.
badgeTextColor = (hex) => {
  const r = parseInt(hex.slice(1,3), 16) / 255;
  const g = parseInt(hex.slice(3,5), 16) / 255;
  const b = parseInt(hex.slice(5,7), 16) / 255;
  const lin = c => c <= 0.04045 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
  const L = 0.2126 * lin(r) + 0.7152 * lin(g) + 0.0722 * lin(b);
  return L > 0.179 ? "#1b1b1b" : "#ffffff";
}

totalStars = allRepos.reduce((n, r) => n + (r.stargazers_count || 0), 0)

orgFollowers = orgData ? (orgData.followers || 0).toLocaleString() : "—"

// Render stats — hides the strip if all three values are missing
{
  const repoCount = orgData ? orgData.public_repos : allRepos.length;
  const pairs = [
    ["stat-repos",      repoCount.toLocaleString()],
    ["stat-stars",      totalStars.toLocaleString()],
    ["stat-followers",  orgFollowers]
  ];
  const allMissing = pairs.every(([, v]) => !v || v === "0" || v === "—");
  const strip = document.querySelector(".phs-stats");
  if (strip) strip.style.display = allMissing ? "none" : "";
  pairs.forEach(([id, val]) => {
    const el = document.querySelector(`#${id} .phs-stat-value`);
    if (el) el.textContent = val;
  });
}

featuredList = FileAttachment("featured.json").json()

// For each featured repo, look it up in allRepos first (already fetched);
// only fall back to an individual API call if it isn't found there.
featuredData = await Promise.all(
  featuredList.map(async (item) => {
    const found = allRepos.find(r => r.name === item.repo);
    if (found) return { ...found, customDescription: item.description };
    const r = await fetch(
      `https://api.github.com/repos/Public-Health-Scotland/${item.repo}`
    );
    const data = r.ok ? await r.json() : { name: item.repo };
    return { ...data, customDescription: item.description };
  })
)

{
  const grid = document.getElementById("featured-grid");

_langStore   = Inputs.input("All")
_searchStore = Inputs.input("")
_sortStore   = Inputs.input("Last updated")
_pageStore   = Inputs.input(0)

// Reactive values from the stores
langFilter  = Generators.input(_langStore)
searchQuery = Generators.input(_searchStore)
sortOrder   = Generators.input(_sortStore)
pageIndex   = Generators.input(_pageStore)

// Render filter controls and repo grid as one reactive section
{
  const PAGE_SIZE = 24;

  // Filter and sort allRepos
  const q = searchQuery.trim().toLowerCase();
  const filtered = allRepos
    .filter(r => {
      const rl = r.language || "Other";
      const matchLang =
        langFilter === "All" ||
        (langFilter === "Other"
          ? !["R", "Python", "Vue", "JavaScript", "TypeScript"].includes(rl)
          : rl === langFilter);
      const matchSearch =
        !q ||
        r.name.toLowerCase().includes(q) ||
        (r.description || "").toLowerCase().includes(q);
      return matchLang && matchSearch;
    })
    .sort((a, b) => {
      if (sortOrder === "Most stars")
        return (b.stargazers_count || 0) - (a.stargazers_count || 0);
      if (sortOrder === "Alphabetical")
        return a.name.localeCompare(b.name);
      return new Date(b.pushed_at) - new Date(a.pushed_at);
    });

  const pageCount = Math.ceil(filtered.length / PAGE_SIZE);
  const page      = Math.min(pageIndex, Math.max(0, pageCount - 1));
  const paged     = filtered.slice(page * PAGE_SIZE, (page + 1) * PAGE_SIZE);

  // Card HTML
  const cardHtml = paged.map(repo => {
    const lang    = repo.language || "Other";
    const colour  = langColour[lang] || "#666666";
    const textCol = badgeTextColor(colour);
    const desc   = (repo.description || "").slice(0, 120);
    const pushed = new Date(repo.pushed_at).toLocaleDateString("en-GB", {
      day: "numeric", month: "short", year: "numeric"
    });
    const stars  = (repo.stargazers_count || 0).toLocaleString();
    return `
      <a href="${repo.html_url}" target="_blank" rel="noopener"
         class="phs-repo-card" aria-label="View ${repo.name} on GitHub">
        <div class="phs-repo-card-top">
          <span class="phs-repo-name">${repo.name}</span>
          <span class="phs-lang-badge" style="background:${colour};color:${textCol}">${lang}</span>
        </div>
        <p class="phs-repo-desc">${desc || "<em>No description</em>"}</p>
        <div class="phs-repo-meta">
          <span aria-label="${stars} stars">★ ${stars}</span>
          <span>Updated ${pushed}</span>
        </div>
      </a>`;
  }).join("");

  // Pagination HTML
  const prevDisabled = page === 0 ? "disabled" : "";
  const nextDisabled = page >= pageCount - 1 ? "disabled" : "";
  const paginationHtml = pageCount > 1 ? `
    <div class="phs-pagination">
      <button class="btn btn-outline-primary btn-sm phs-page-btn" ${prevDisabled}
        data-page="${page - 1}">← Previous</button>
      <span class="phs-page-info">Page ${page + 1} of ${pageCount}</span>
      <button class="btn btn-outline-primary btn-sm phs-page-btn" ${nextDisabled}
        data-page="${page + 1}">Next →</button>
    </div>` : "";

  // Error / empty state
  const emptyHtml = filtered.length === 0
    ? `<p class="phs-repo-empty">No repositories match your filters.
       <a href="https://github.com/Public-Health-Scotland">View all on GitHub →</a></p>`
    : "";

  // Inject
  const container = document.getElementById("repo-section-body");
  if (!container) return;

  container.innerHTML = `
    <p class="phs-repo-count">${
      filtered.length === 0
        ? ""
        : `Showing ${paged.length} of ${filtered.length} repositories`
    }</p>
    <div class="phs-repo-grid">${emptyHtml || cardHtml}</div>
    ${paginationHtml}`;

  // Wire pagination buttons
  container.querySelectorAll(".phs-page-btn:not([disabled])").forEach(btn => {
    btn.addEventListener("click", () => {
      _pageStore.value = +btn.dataset.page;
      _pageStore.dispatchEvent(new Event("input"));
    });
  });
}
{
  langFilter; searchQuery; sortOrder;
  _pageStore.value = 0;
  _pageStore.dispatchEvent(new Event("input"));
}
// Render filter controls into the placeholder div.
// Inputs.bind creates a new visible control bound to the backing store.
{
  const container = document.getElementById("repo-controls");
  if (!container) return;

  const langInput = Inputs.bind(
    Inputs.radio(["All", "R", "Python", "Vue", "Other"], {
      label: "Language", value: "All"
    }),
    _langStore
  );
  const searchInput = Inputs.bind(
    Inputs.text({ label: "Search", placeholder: "Filter by name or description…" }),
    _searchStore
  );
  const sortInput = Inputs.bind(
    Inputs.select(["Last updated", "Most stars", "Alphabetical"], { label: "Sort by" }),
    _sortStore
  );

  container.innerHTML = "";
  container.append(langInput, searchInput, sortInput);
}
